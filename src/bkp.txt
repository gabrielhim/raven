#[derive(Parser)]
struct AnnotateArgs {
    /// Reference VCF
    #[arg(short, long)]
    dataset: String,

    /// Input VCF
    #[arg(short, long)]
    vcf: String,

    /// Output format
    #[arg(long, short = 'f', value_enum, default_value_t = OutputFormat::Json)]
    output_format: OutputFormat,

    /// Output prefix
    #[arg(long, short)]
    output_prefix: String,
}

enum AnnotationType {
    ScalarFloat(f32),
    ScalarInteger(i32),
    ScalarString(String),
    VecFloat(Vec<f32>),
    VecInteger(Vec<i32>),
    VecString(Vec<String>),
}

struct TagAnnotation {
    tag: String,
    annot_type: AnnotationType,
}

struct VariantAnnotation<'a> {
    variant: Variant<'a>,
    annotations: Vec<TagAnnotation>,
}

impl<'a> VariantAnnotation<'a> {
    fn new(variant: Variant<'a>) -> Self {
        Self {
            variant,
            annotations: Vec::new(),
        }
    }
}
